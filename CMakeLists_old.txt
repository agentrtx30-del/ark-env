cmake_minimum_required(VERSION 3.15)
project(ark_env CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(ark_env ark_env.cpp)
target_compile_definitions(ark_env PRIVATE UNICODE _UNICODE)

if (MSVC)
    target_compile_options(ark_env PRIVATE /W4)
    target_link_libraries(ark_env PRIVATE psapi version advapi32 user32)
endif()
